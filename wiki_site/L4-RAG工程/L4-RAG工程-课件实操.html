RAG 第一步永远是把长文档切成小块（chunk）。课件从"按字符切 → 滑动窗口 → 递归切"三步演进，最终推荐递归分割器：

# 来源：day01/源码/01-文档分割.py
from langchain_text_splitters import RecursiveCharacterTextSplitter

text = "自然语言处理（NLP），作为计算机科学、人工智能与语言学的交融之地……"

# chunk_overlap 防止语义冲突：相邻 chunk 重叠一部分，避免关键句被切两半
split = RecursiveCharacterTextSplitter(
    chunk_size=100,
    chunk_overlap=10,
    separators=["\n", ""],
)
chunks = split.split_text(text)
for i, chunk in enumerate(chunks):
    print(f"块 {i + 1}: {len(chunk)}: {chunk}")

参数怎么调（day06 笔记总结）：
- chunk_size：256–512 是甜点区。医疗/法律/生物等严格场景要切大一点（1024+），社媒/短文本切小（200）
- chunk_overlap：通常设为 chunk_size 的 10%–20%（如 256 的块设 25–50）
- separators：优先级从高到低排列，先按段落切、再按句号、再按空格，是"递归"含义所在

常见坑：
- 不要用固定字符硬切（如每 250 字一刀），会把句子切烂导致 embedding 语义断裂
- 表格、代码块要单独处理（用 MarkdownHeaderTextSplitter 或 HTMLHeaderTextSplitter），不能走通用分割

# 来源：day01/源码/05-向量维度计算.py
import numpy as np
from numpy import dot
from numpy.linalg import norm

def cos_sim(a, b):
    '''余弦相似度 —— 越大越相似（0~1）'''
    return dot(a, b) / (norm(a) * norm(b))

def l2(a, b):
    '''欧氏距离 —— 越小越相似'''
    x = np.asarray(a) - np.asarray(b)
    return norm(x)

# 实际使用：对 query_vecs 和 doc_vecs 逐个打分，取 top-k
# 余弦相似度：适合场景复杂、长短文本混合
# 欧氏距离：适合向量已经归一化的场景

# 来源：day02/源码/04-实现rag项目.py
import chromadb
from pdfminer.high_level import extract_pages
from pdfminer.layout import LTTextContainer
from base_llm import client  # OpenAI client

# ① 切分：固定字符 + 滑动窗口
def sliding_window_chunks(text, chunk_size, stride):
    return [text[i:i + chunk_size] for i in range(0, len(text), stride)]

# ② PDF 提取
def extract_text_from_pdf(filename, page_numbers=None):
    full_text = ''
    for i, page_layout in enumerate(extract_pages(filename)):
        if page_numbers is not None and i not in page_numbers:
            continue
        for element in page_layout:
            if isinstance(element, LTTextContainer):
                # 去掉换行和空格（中文 PDF 常见换行垃圾）
                full_text += element.get_text().replace("\n", "").replace(" ", "")
    return sliding_window_chunks(full_text, 250, 100)

# ③ 向量库封装
class MyVectorDBConnector:
    def __init__(self, collection_name):
        chroma_client = chromadb.Client()
        self.collection = chroma_client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"},  # 关键：切到余弦相似度
        )

    def get_embeddings(self, texts):
        data = client.embeddings.create(input=texts, model="text-embedding-v2")
        return [x.embedding for x in data.data]

    def add_documents(self, data):
        emb = self.get_embeddings(data)
        self.collection.add(
            embeddings=emb,
            documents=data,
            ids=[f"id{i}" for i in range(len(data))],
        )

    def search(self, query, n_results):
        return self.collection.query(
            query_embeddings=self.get_embeddings([query]),
            n_results=n_results,
        )

# ④ RAG 机器人
class RAG_Bot:
    def __init__(self, vector_db, n_results=2):
        self.vector_db = vector_db
        self.n_results = n_results

    def get_llm(self, prompt):
        messages = [{"role": "user", "content": prompt}]
        response = client.chat.completions.create(model='qwen3-max', messages=messages)
        return response.choices[0].message.content

    def chat(self, user_query):
        # 检索
        search_res = self.vector_db.search(user_query, n_results=self.n_results)
        # 拼接 prompt（模板里两个占位符）
        prompt = prompt_template.replace('__INFO__', '\n'.join(search_res['documents'][0])) \
                                .replace('__QUERY__', user_query)
        # 生成
        print(self.get_llm(prompt))

# ⑤ 运行
if __name__ == '__main__':
    prompt_template = """
        你是一个问答机器人。
        你的任务是根据下述给定的已知信息回答用户问题。
        确保你的回复完全依据下述已知信息。不要编造答案。
        如果下述已知信息不足以回答用户的问题，请直接回复"我无法回答您的问题"。

        已知信息:
        __INFO__

        用户问:
        __QUERY__

        请用中文回答用户问题。
    """
    data = extract_text_from_pdf('财务管理文档.pdf', page_numbers=[0, 1, 2])
    vect = MyVectorDBConnector('test')
    vect.add_documents(data)

    rag = RAG_Bot(vect)
    rag.chat("财务管理权限划分?")

这个最小版的三个核心设计（也是所有 RAG 的骨架）：
1. 索引期：PDF → 文本 → 滑动窗口切 250 字 → embedding → Chroma
2. 检索期：用户问题 → embedding → Chroma top-k=2 → 拿到 docs
3. 生成期：docs + 问题 → 填 prompt 模板 → LLM

Prompt 模板的"反幻觉三句话"："完全依据已知信息 / 不要编造 / 不足以回答就说无法回答" —— 这是最经济的防幻觉手段，生产环境务必保留。

2.2 带历史的会话链（day06 核心）

3 
# 来源：day06/源码/03-实现记忆功能.py
import json
from langchain_core.runnables.history import RunnableWithMessageHistory
from langchain_community.chat_message_histories import ChatMessageHistory
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.messages import messages_from_dict, messages_to_dict
from base_llm import llm

# ① Prompt 模板：占位符 {history} 是给历史消息留的位置
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个友好的助手"),
    MessagesPlaceholder(variable_name="history"),   # 历史消息槽
    ("user", "{input}"),                            # 当前用户输入槽
])

# ② 基础链：prompt | llm
base_chain = prompt | llm

# ③ 内存里的会话字典：{session_id: ChatMessageHistory}
store = {}
def get_session_history(session_id):
    if session_id not in store:
        store[session_id] = ChatMessageHistory()
    return store[session_id]

# ④ 用 RunnableWithMessageHistory 包一层：自动读/写历史
conv = RunnableWithMessageHistory(
    base_chain,
    get_session_history=get_session_history,
    input_messages_key='input',
    history_messages_key='history',
)

def predict(input_text, session_id="default"):
    return conv.invoke(
        {"input": input_text},
        config={"configurable": {"session_id": session_id}},  # 关键：session_id 关联历史
    ).content

# ⑤ 持久化：把历史 dump 成 JSON
def save_memory(filepath, session_id):
    history = get_session_history(session_id)
    dicts = messages_to_dict(history.messages)
    with open(filepath, "w", encoding='utf-8') as f:
        json.dump(dicts, f, ensure_ascii=False)

def load_memory(filepath, session_id):
    with open(filepath, "r", encoding='utf-8') as f:
        dicts = json.load(f)
    messages = messages_from_dict(dicts)
    store[session_id] = ChatMessageHistory(messages=messages)

# ⑥ 测试
predict("你好,我是柏汌")
predict("我叫什么名字?")  # 能正确回答"柏汌"
save_memory("history.json", "default")

load_memory("history.json", "default")
print(predict("我回来了，我们之前都聊了一些什么?"))

关键参数：
- input_messages_key='input'：告诉链"用户当前输入"在哪个变量
- history_messages_key='history'：对应 prompt 里 MessagesPlaceholder 的 variable_name
- session_id：多会话隔离的钥匙，同一 session_id 才共享历史

常见坑：
- 历史太长会爆 token（课件笔记：10–30 轮就该压缩或截断）
- 生产用 RedisChatMessageHistory / SQLChatMessageHistory，不要用内存字典，重启就丢

# 来源：day07/源码/01-摘要索引.py
from base_llm import llm, embeddings_model
from langchain_core.stores import InMemoryByteStore
from langchain_chroma import Chroma
from langchain_community.document_loaders import UnstructuredWordDocumentLoader, WebBaseLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_classic.retrievers import MultiVectorRetriever
from langchain_core.documents import Document
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
import uuid

# ① 加载：Word 文档 + 网页
loaders = [
    UnstructuredWordDocumentLoader("人事管理流程.docx"),
    WebBaseLoader("https://news.pku.edu.cn/mtbdnew/15ac0b3e79244efa88b03a570cbcbcaa.htm"),
]
docs = []
for loader in loaders:
    docs.extend(loader.load())

# ② 切分
text_split = RecursiveCharacterTextSplitter(chunk_size=1024, chunk_overlap=100)
docs = text_split.split_documents(docs)

# ③ 批量生成摘要（并发 5）
chain = (
    {"doc": lambda x: x.page_content}
    | ChatPromptTemplate.from_template("总结下面的文档:\n\n{doc}")
    | llm
    | StrOutputParser()
)
summ = chain.batch(docs, {'max_concurrency': 5})

# ④ 双存储：向量库存摘要 + ByteStore 存原文
vector = Chroma(collection_name='summ', embedding_function=embeddings_model)
store = InMemoryByteStore()
id_key = 'doc_id'

retriever = MultiVectorRetriever(
    vectorstore=vector,
    byte_store=store,
    id_key=id_key,
)

# ⑤ 给每个原文档一个 uuid，把摘要和 id 一起入向量库
doc_ids = [str(uuid.uuid4()) for _ in docs]
summary_docs = [
    Document(page_content=s, metadata={id_key: doc_ids[i]})
    for i, s in enumerate(summ)
]
retriever.vectorstore.add_documents(summary_docs)
retriever.docstore.mset(list(zip(doc_ids, docs)))

# ⑥ 使用：向量库命中摘要 → 自动通过 metadata 的 doc_id 拿回原文
sub_docs = retriever.vectorstore.similarity_search("病假的请假流程?")
orig_doc = retriever.docstore.mget([sub_docs[0].metadata[id_key]])
print(orig_doc)

为什么不直接把原文向量化？ 课件笔记原话："摘要语义聚焦，向量空间更集中，召回率高；原文太长 embedding 会被稀释，短查询命中不准。"

from langchain_core.stores import InMemoryByteStore
from langchain_chroma import Chroma
from langchain_community.document_loaders import TextLoader

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnableMap

import uuid

""")

doc_ids = [str(uuid.uuid4()) for _ in docs]

question_docs = []
for i, q_list in enumerate(hypothetical_questions):
    question_docs.extend(
        [Document(page_content=q, metadata={id_key: doc_ids[i]}) for q in q_list]

retriever.vectorstore.add_documents(question_docs)
retriever.docstore.mset(list(zip(doc_ids, docs)))

    prompt | llm | StrOutputParser() | (lambda x: x.split("\n"))
)

original_query = "人工智能的应用"
queries = generate_queries.invoke({"original_query": original_query})
print(queries)  # ['人工智能医疗应用', 'AI在金融领域', ...]

# ③ RRF 算法核心：对多个排序列表，按"1/(rank+k)"累加分数
def reciprocal_rank_fusion(results: list[list], k=60):
    fused_scores = {}
    for docs in results:
        for rank, doc in enumerate(docs):
            doc_str = dumps(doc)  # 序列化做唯一标识
            if doc_str not in fused_scores:
                fused_scores[doc_str] = 0
            fused_scores[doc_str] += 1 / (rank + k)  # k=60 平滑参数

    sorted_data = sorted(fused_scores.items(), key=lambda x: x[1], reverse=True)
    return [(loads(doc), score) for doc, score in sorted_data]

# ④ 组装：生成查询 → 并行检索 → RRF 融合
chain = generate_queries | retriever.map() | reciprocal_rank_fusion
result_list = chain.invoke({"original_query": original_query})

for doc, score in result_list:
    print(doc.page_content, score)

RRF 为什么用 k=60：经验值。文档排名越靠前加分越多，但排名第 1 的文档不会因为出现在多个列表里就无脑霸榜（1/(0+60) vs 1/(1+60) 差距很小）。这让结果更鲁棒。

# ③ 单 chunk 抽取 → GraphDocument 对象
def extract_one_document(doc: Document) -> GraphDocument:
    raw = extract_chain.invoke({"text": doc.page_content})
    if not isinstance(raw, dict):
        return GraphDocument(nodes=[], relationships=[], source=doc)

    nodes, node_map = [], {}
    for n in raw.get("nodes", []):
        nid = n.get("id")
        if not nid or nid in node_map:
            continue
        node = LCNode(id=nid, type=n.get("type", "Entity"), properties=n.get("properties", {}))
        node_map[nid] = node
        nodes.append(node)

    relationships = []
    for r in raw.get("relationships", []):
        s, t, rt = r.get("source"), r.get("target"), r.get("type")
        if not (s and t and rt) or s not in node_map or t not in node_map:
            continue
        relationships.append(LCRel(
            source=node_map[s], target=node_map[t],
            type=rt, properties=r.get("properties", {}),
        ))

    return GraphDocument(nodes=nodes, relationships=relationships, source=doc)

def ingest_data(file_path: str):
    with open(file_path, "r", encoding="utf-8") as f:
        text = f.read()
    chunks = RecursiveCharacterTextSplitter(
        chunk_size=1200, chunk_overlap=200,
        separators=["\n\n", "\n", "。", "，", " ", ""],
    ).split_text(text)
    documents = [Document(page_content=c) for c in chunks]

    graph_documents = []
    for i, doc in enumerate(documents, 1):
        gd = extract_one_document(doc)
        if gd.nodes or gd.relationships:
            graph_documents.append(gd)

    # ⑤ 写入 Neo4j
    graph.add_graph_documents(
        graph_documents,
        baseEntityLabel=True,   # 每个实体打 __Entity__ 标签
        include_source=True,   # 把原始 chunk 也存成节点，用 MENTIONS 关系连实体
    )

def get_graph_rag_chain():
    graph.refresh_schema()
    chain = GraphCypherQAChain.from_llm(
        llm=llm,
        graph=graph,
        verbose=True,
        allow_dangerous_requests=True,
        top_k=30,
    )

这个链路的本质：
- 建图：文档 → LLM 抽 (实体, 关系, 实体) 三元组 → 写入 Neo4j 节点和边
- 查询：用户自然语言问题 → LLM 翻译成 Cypher → Neo4j 执行 → 结果转回自然语言

GraphRAG vs 向量 RAG：
- 向量 RAG：擅长"某段文字讲了什么"（语义匹配）
- GraphRAG：擅长"实体之间的关系是什么"（多跳推理，如 A 的师父的师父是谁）
- 据[一手源]：RAG图谱行业标准调研-20261009（§3）——课件只讲了 GraphRAG 的长处，反面实测要补上：简单事实上 GraphRAG 比 vanilla RAG 低 13.4%（Natural Questions，转引 Han et al. 2025）；HotpotQA 多跳仅 +4.5% 但延迟 2.3×。跑下面 Neo4j 链路前先问：真实 query 里多跳占多少，单点事实为主就别上图谱。

from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.document_loaders import PyPDFLoader, TextLoader, Docx2txtLoader
import os, tempfile

class DocumentProcessor:
    def __init__(self, config):
        self.config = config

        # 父分割器：大块（2倍 chunk_size），保留语义完整性
        self.parent_splitter = RecursiveCharacterTextSplitter(
            chunk_size=config.CHUNK_SIZE * 2,
            chunk_overlap=config.CHUNK_OVERLAP,
            separators=["\n\n", "\n", "。", "！", "？"],
        )
        # 子分割器：小块（1/2 chunk_size），用于精确向量匹配
        self.child_splitter = RecursiveCharacterTextSplitter(
            chunk_size=config.CHUNK_SIZE // 2,
            chunk_overlap=config.CHUNK_OVERLAP // 2,
            separators=["\n", "。", "！", "？", ".", "!", "?", " "],
        )

    def load_document(self, uploaded_file):
        # Streamlit 上传的文件先落到临时盘
        with tempfile.NamedTemporaryFile(delete=False, suffix=f".{uploaded_file.name.split('.')[-1]}") as tmp:
            tmp.write(uploaded_file.getvalue())
            tmp_path = tmp.name
        try:
            if uploaded_file.name.endswith('.pdf'):
                loader = PyPDFLoader(tmp_path)
            elif uploaded_file.name.endswith('.docx'):
                loader = Docx2txtLoader(tmp_path)
            elif uploaded_file.name.endswith('.txt'):
                loader = TextLoader(tmp_path, encoding='utf-8')
            else:
                raise ValueError(f"不支持的文件类型: {uploaded_file.name}")
            return loader.load()
        finally:
            os.unlink(tmp_path)

    def create_parent_child_chunks(self, documents, fileName):
        parent_docs = self.parent_splitter.split_documents(documents)
        child_docs = []
        for i, parent_doc in enumerate(parent_docs):
            parent_id = f"{fileName}_parent_{i}"
            parent_doc.metadata['parent_id'] = parent_id
            parent_doc.metadata['doc_type'] = 'parent'

            # 从每个父文档再切出子文档
            child_chunks = self.child_splitter.split_documents([parent_doc])
            for j, child_doc in enumerate(child_chunks):
                child_doc.metadata['parent_id'] = parent_id
                child_doc.metadata['child_id'] = f"child_{i}_{j}"
                child_doc.metadata['doc_type'] = 'child'
                child_docs.append(child_doc)
        return parent_docs, child_docs

设计要点：

- 子分割器 separator 到句子级，保证子块向量聚焦
- 每个 chunk 带 parent_id 元数据，后续可双向追溯

import chromadb
from langchain_chroma import Chroma
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_classic.retrievers import ContextualCompressionRetriever
from langchain_classic.retrievers.document_compressors import LLMChainExtractor
from langchain_openai import ChatOpenAI

class VectorStore:
    def __init__(self, config):
        self.config = config
        self.embeddings = HuggingFaceEmbeddings(
            model_name=r"D:\llm\Local_model\BAAI\bge-large-zh-v1___5"
        )
        self.chroma_client = chromadb.PersistentClient(path=config.CHROMA_PERSIST_DIR)

        # 双集合：父、子分开存
        self.parent_vectorstore = Chroma(
            client=self.chroma_client, collection_name="parent_documents",
            embedding_function=self.embeddings,
        )
        self.child_vectorstore = Chroma(
            client=self.chroma_client, collection_name="child_documents",
            embedding_function=self.embeddings,
        )

        # 上下文压缩器：检索后让 LLM 从每个 chunk 里只抽取相关句子
        self.llm = ChatOpenAI(
            api_key=config.OPENAI_API_KEY,
            base_url=config.OPENAI_BASE_URL,
            model="qwen-plus-2025-04-28",
            temperature=0,
        )
        self.compressor = LLMChainExtractor.from_llm(self.llm)

    def add_documents(self, parent_docs, child_docs, document_id):
        for doc in parent_docs + child_docs:
            doc.metadata['document_id'] = str(document_id)
        parent_ids = self.parent_vectorstore.add_documents(parent_docs)
        child_ids = self.child_vectorstore.add_documents(child_docs)
        return parent_ids, child_ids

    def create_retriever(self, use_compression=True, document_ids=None):
        search_kwargs = {"k": self.config.TOP_K * 2}
        # 按用户勾选的文档过滤（多文档隔离）
        if document_ids:
            search_kwargs["filter"] = {"document_id": {"$in": [str(id) for id in document_ids]}}

        child_retriever = self.child_vectorstore.as_retriever(search_kwargs=search_kwargs)

        if use_compression:
            return ContextualCompressionRetriever(
                base_compressor=self.compressor,
                base_retriever=child_retriever,
            )
        return child_retriever

    def get_parent_documents(self, child_docs):
        # 用子文档的 parent_id 元数据，回父向量库取对应父块
        parent_ids = {doc.metadata['parent_id'] for doc in child_docs if 'parent_id' in doc.metadata}
        if not parent_ids:
            return []
        parent_docs = []
        for pid in parent_ids:
            results = self.parent_vectorstore.get(where={"parent_id": pid})
            if results['documents']:
                parent_docs.append(results['documents'][0])
        return parent_docs

生产级设计要点：
1. 父子双库：子库做精确匹配（k=10），父库提供完整上下文
2. document_id 过滤：用户勾选哪些文档就只在哪些文档里检索，多租户隔离
3. ContextualCompressionRetriever：先召回 10 个 child chunk，让 LLM 从每个 chunk 里抽相关句，再喂给生成模型。token 消耗降 60%，答案更聚焦

from langchain_openai import ChatOpenAI
from langchain_core.prompts import PromptTemplate
import uuid

class RAGSystem:
    def __init__(self, config, db_manager, vector_store):
        self.config = config
        self.db_manager = db_manager
        self.vector_store = vector_store

        self.llm = ChatOpenAI(
            api_key=config.OPENAI_API_KEY,
            base_url=config.OPENAI_BASE_URL,
            model="qwen-plus-2025-04-28",
            temperature=0.7,
            streaming=True,
        )

        # RAG Prompt：三件套（上下文 + 历史 + 问题）
        self.rag_prompt = PromptTemplate(
            template="""你是一个智能助手。请基于以下上下文信息回答用户的问题。如果上下文中没有相关信息，请诚实地说明。

上下文信息:
{context}

历史对话:
{chat_history}

用户问题: {question}

请提供准确、有帮助的回答:""",
            input_variables=["context", "chat_history", "question"],
        )

    def generate_session_id(self):
        return str(uuid.uuid4())

    def chat_with_documents(self, question, document_ids, session_id):
        try:
            # ① 检索（带子文档压缩）
            retriever = self.vector_store.create_retriever(
                use_compression=True, document_ids=document_ids,
            )
            retrieved_docs = retriever.invoke(question)

            # ② 用子文档的 parent_id 拿回父文档（提供完整上下文）
            parent_docs = self.vector_store.get_parent_documents(retrieved_docs)
            context = "\n\n".join(parent_docs)

            # ③ 历史只取最近 3 轮（控 token）
            chat_history = self.db_manager.get_chat_history(session_id)
            history_text = "\n".join([
                f"用户: {c.user_message}\n助手: {c.assistant_message}"
                for c in chat_history[-3:]
            ])

            # ④ 填 prompt 生成
            full_prompt = self.rag_prompt.format(
                context=context, chat_history=history_text, question=question,
            )
            answer = self.llm.invoke(full_prompt).content

            # ⑤ 存历史
            self.db_manager.save_chat_history(
                session_id=session_id,
                user_message=question,
                assistant_message=answer,
                document_ids=",".join(map(str, document_ids)),
            )
            return answer, retrieved_docs

        except Exception as e:
            return f"RAG对话出错: {e}", []

    def normal_chat(self, question, session_id):
        # 无文档的纯对话，历史取最近 5 轮
        chat_history = self.db_manager.get_chat_history(session_id)
        messages = []
        for c in chat_history[-5:]:
            messages.append(f"Human: {c.user_message}")
            messages.append(f"Assistant: {c.assistant_message}")
        messages.append(f"Human: {question}")
        response = self.llm.invoke("\n".join(messages) + "\nAssistant:")
        self.db_manager.save_chat_history(session_id, question, response.content)
        return response.content

Base = declarative_base()

class Document(Base):
    __tablename__ = 'documents'
    id = Column(Integer, primary_key=True, autoincrement=True)
    filename = Column(String(255), nullable=False)
    content = Column(MEDIUMTEXT, nullable=False)
    chunk_count = Column(Integer, default=0)
    created_at = Column(DateTime, default=datetime.now())
    is_active = Column(Boolean, default=True)

    parent_chunks = relationship("ParentChunk", back_populates="document",
                                  cascade="all, delete-orphan")
    child_chunks = relationship("ChildChunk", back_populates="document",
                                 cascade="all, delete-orphan")

class ParentChunk(Base):
    __tablename__ = 'parent_chunks'
    id = Column(Integer, primary_key=True, autoincrement=True)
    document_id = Column(Integer, ForeignKey('documents.id'))
    parent_id = Column(String(100), unique=True)
    content = Column(Text, nullable=False)
    vector_id = Column(String(100))   # 对应 Chroma 中的向量 ID

    document = relationship("Document", back_populates="parent_chunks")
    child_chunks = relationship("ChildChunk", back_populates="parent_chunk",
                                cascade="all, delete-orphan")

    __table_args__ = (
        Index('idx_parent_document_id', 'document_id'),
        Index('idx_parent_id', 'parent_id'),
    )

class ChildChunk(Base):
    __tablename__ = 'child_chunks'
    id = Column(Integer, primary_key=True, autoincrement=True)
    document_id = Column(Integer, ForeignKey('documents.id'))
    parent_chunk_id = Column(Integer, ForeignKey('parent_chunks.id'))
    child_id = Column(String(100))
    content = Column(Text)
    vector_id = Column(String(100))

    document = relationship("Document", back_populates="child_chunks")
    parent_chunk = relationship("ParentChunk", back_populates="child_chunks")

class ChatHistory(Base):
    __tablename__ = 'chat_history'
    id = Column(Integer, primary_key=True, autoincrement=True)
    session_id = Column(String(64), index=True)
    user_message = Column(Text)
    assistant_message = Column(Text)
    document_ids = Column(String(255))
    created_at = Column(DateTime, default=datetime.now())
