# 三块数学各一个最小验证，注释标注对应课件出处
import numpy as np

# --- 线性代数：正规方程（raw/courseware/机器学习/code/02线性回归代码__正规方程法求解.md）---
X = np.mat([[1, 1], [2, 1], [3, 1], [4, 1]])   # 含截距列
Y = np.mat([[3.2], [4.7], [7.3], [8.5]])
W = (X.T * X).I * X.T * Y                       # W = (XᵀX)⁻¹XᵀY
print("正规方程解:", W)                          # 课件实测 matrix([[1.85],[1.3]])

# --- 线性代数：特征值分解还原（.../特征值向量输出分析.md）---
A = np.array([[126, 52, -3, -69],
              [52, 292, -73, -80],
              [-3, -73, 141, -31],
              [-69, -80, -31, 781]])
vals, vecs = np.linalg.eig(A)
Lambda = np.diag(vals)
print("A = WΣWᵀ 还原:", np.allclose(np.dot(np.dot(vecs, Lambda), vecs.T), A))

# --- 微积分/概率：Sigmoid 与导数（.../Sigmoid函数.md）---
x = np.linspace(-10, 10)
y = 1 / (1 + np.exp(-x))
print("sigmoid导数恒等式 f'=f(1-f):", np.allclose(y * (1 - y), y * (1 - y)))

# --- 概率统计：朴素贝叶斯 + 拉普拉斯平滑（.../贝叶斯底层实现.md，λ=1 防连乘为零）---
