# BM25 + 向量 + 重排序
def hybrid_search(query, top_k=10):
    # 1. BM25检索
    bm25_results = bm25_search(query, top_k * 2)

    # 2. 向量检索
    vector_results = vector_search(query, top_k * 2)

    # 3. 融合（RRF）
    merged = reciprocal_rank_fusion(bm25_results, vector_results)

    # 4. 重排序
    reranked = reranker.rank(query, merged[:top_k * 2])

    return reranked[:top_k]

5.2 关键设计原则

3

1. 模型无关性

# 抽象模型接口
class LLMProvider:
    def chat(messages, tools=None) -> Response
    def embed(texts) -> List[float]

2. 可观测性

# 追踪每次调用
@trace
def rag_query(question):
    docs = retrieve(question)
    answer = generate(question, docs)
    return answer

3. 缓存策略

# 语义缓存
cache = SemanticCache(threshold=0.95)
if cached := cache.get(query):
    return cached
result = llm(query)
cache.set(query, result)

4. 降级策略

# 主模型失败自动降级
providers = [primary, secondary, fallback]
for provider in providers:
    try:
        return provider.chat(messages)
    except:
        continue
